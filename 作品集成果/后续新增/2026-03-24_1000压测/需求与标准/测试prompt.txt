FROM_SYNONYMS = ["从", "由", "自"]
TO_SYNONYMS = ["到", "去", "前往", "走到"]
FILLER_WORDS = ["啊", "那个", "嗯", "就是"]

def augment_case(case):
    text = case["input"]
    augmented = case.copy()
    if random.random() > 0.5:
        for word in FROM_SYNONYMS:
            if word in text:
                text = text.replace(word, random.choice(FROM_SYNONYMS), 1)
                break
    if random.random() > 0.5:
        for word in TO_SYNONYMS:
            if word in text:
                text = text.replace(word, random.choice(TO_SYNONYMS), 1)
                break
    if random.random() > 0.6:
        insert_pos = random.randint(0, len(text))
        text = text[:insert_pos] + random.choice(FILLER_WORDS) + text[insert_pos:]
    if random.random() > 0.5:
        text = text.replace("，", "").replace("。", "").replace("？", "")
    augmented["input"] = text
    return augmented

def build_test_suite(target_count=1000):
    suite = list(SEED_CASES)
    random.seed(42)
    while len(suite) < target_count:
        suite.append(augment_case(random.choice(SEED_CASES)))
    return suite[:target_count]

# 2. 基线对照组（模拟直接数据库查询，无容错能力）
# ============================================================
def baseline_exact_match(user_input, place_names):
    """模拟没有算法的直接字符串匹配（基线对照组）"""
    for name in place_names:
        if name in user_input:
            return name
    return None

def evaluate_baseline(test_cases):
    """统计基线准确率"""
    success = 0
    for case in test_cases:
        if not case.get("has_nav_intent"):
            success += 1
            continue
        expected_end = case.get("expected_end")
        if expected_end is None:
            success += 1
            continue
        result = baseline_exact_match(case["input"], PLACE_NAMES)
        if result and case["expected_end"][:2] in (result or ""):
            success += 1
    return success / len(test_cases)

# 3. 评估函数
# ============================================================
def evaluate_result(test_case, response_json, latency_ms):
    if response_json is None:
        return False, "TIMEOUT", None
    debug = response_json.get("debug_info", {})
    actual_end = debug.get("matched_end_place") or ""
    actual_start = debug.get("matched_start_place") or ""
    actual_nav_intent = debug.get("navigation_intent", False)
    match_method = debug.get("match_method")

    expected_end = test_case.get("expected_end")
    expected_start = test_case.get("expected_start")
    has_nav_intent = test_case.get("has_nav_intent", True)

    if not has_nav_intent:
        if actual_nav_intent:
            return False, "FALSE_NAV_TRIGGER", match_method
        return True, None, "non_nav"

    if has_nav_intent and not actual_nav_intent and expected_end is not None:
        return False, "MISSED_NAV_INTENT", match_method

    if expected_end is not None:
        if expected_end[:2] not in actual_end:
            return False, "END_MISMATCH", match_method

    if expected_start is not None:
        if expected_start[:2] not in actual_start:
            return False, "START_MISMATCH", match_method

    return True, None, match_method

# 4. 异步请求
# ============================================================
async def send_one_request(session, test_case, semaphore, url):
    async with semaphore:
        start_time = time.perf_counter()
        result = {
            "input": test_case["input"],
            "scene": test_case.get("scene", "unknown"),
            "match_level_expected": test_case.get("match_level_expected"),
            "expected_start": test_case.get("expected_start"),
            "expected_end": test_case.get("expected_end"),
            "has_nav_intent": test_case.get("has_nav_intent", True),
            "latency_ms": 0,
            "http_status": None,
            "response_json": None,
            "is_success": False,
            "fail_reason": None,
            "match_method_actual": None,
        }
        try:
            timeout = aiohttp.ClientTimeout(total=8)
            payload = {"message": test_case["input"], "test_mode": True,
                       "strategy": "safest", "asr_confidence": None, "user_bearing": None}
            async with session.post(url, json=payload, timeout=timeout) as resp:
                result["http_status"] = resp.status
                result["latency_ms"] = round((time.perf_counter() - start_time) * 1000, 2)
                if resp.status == 200:
                    result["response_json"] = await resp.json()
                else:
                    result["fail_reason"] = "HTTP_ERROR"
                    return result
        except asyncio.TimeoutError:
            result["latency_ms"] = round((time.perf_counter() - start_time) * 1000, 2)
            result["fail_reason"] = "TIMEOUT"
            return result
        except aiohttp.ClientConnectorError:
            print("\n❌ 无法连接到后端！请先执行: uvicorn main:app --reload")
            raise SystemExit(1)

        is_success, fail_reason, match_method = evaluate_result(
            test_case, result["response_json"], result["latency_ms"])
        result["is_success"] = is_success
        result["fail_reason"] = fail_reason
        result["match_method_actual"] = match_method
        return result

# ============================================================
# 5. 主压测
# ============================================================
async def run_stress_test(url, concurrency, count):
    test_suite = build_test_suite(count)
    semaphore = asyncio.Semaphore(concurrency)
    results = []
    print(f"\n{'='*60}")
    print(f"🚀 鼓浪屿导航系统压测启动 | 并发:{concurrency} | 总数:{count}")
    print(f"{'='*60}")
    connector = aiohttp.TCPConnector(limit=concurrency)
    async with aiohttp.ClientSession(connector=connector) as session:
        tasks = [send_one_request(session, case, semaphore, url) for case in test_suite]
        completed = 0
        for coro in asyncio.as_completed(tasks):
            result = await coro
            results.append(result)
            completed += 1
            if completed % 100 == 0 or completed == count:
                rate = sum(1 for r in results if r["is_success"]) / completed * 100
                print(f"  进度: {completed}/{count} | 实时准确率: {rate:.1f}%")
    return results, test_suite

# 7. 报告生成
# ============================================================
def generate_report(results, test_suite, output_dir):
    total = len(results)
    http_success = sum(1 for r in results if r["http_status"] == 200)
    algo_success = sum(1 for r in results if r["is_success"])
    timeouts = sum(1 for r in results if r["fail_reason"] == "TIMEOUT")

    latencies = sorted([r["latency_ms"] for r in results if r["http_status"] == 200 and r["latency_ms"] > 0])

    def pct(data, p):
        if not data: return 0
        return data[min(int(len(data) * p / 100), len(data)-1)]

    avg_lat = sum(latencies) / len(latencies) if latencies else 0
    under_100 = sum(1 for l in latencies if l <= 100) / len(latencies) * 100 if latencies else 0
    under_500 = sum(1 for l in latencies if l <= 500) / len(latencies) * 100 if latencies else 0
    baseline_rate = evaluate_baseline(test_suite) * 100

    scene_stats = defaultdict(lambda: {"total": 0, "success": 0})
    fail_reasons = defaultdict(int)
    for r in results:
        scene_stats[r["scene"]]["total"] += 1
        if r["is_success"]:
            scene_stats[r["scene"]]["success"] += 1
        if not r["is_success"] and r["fail_reason"]:
            fail_reasons[r["fail_reason"]] += 1

    accuracy_rate = algo_success / total
    improvement = accuracy_rate * 100 - baseline_rate

    lines = [
        "=" * 62,
        "  鼓浪屿导航系统 | 1000条压测完整报告",
        f"  生成时间：{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}",
        "=" * 62,
        "",
        "【核心指标对照 SOP 验收标准】",
        f"  意图识别准确率:    {accuracy_rate*100:.1f}%   {'✅ 达标' if accuracy_rate >= 0.85 else '❌ 未达标'}  (SOP要求 ≥85%)",
        f"  平均响应时间:      {avg_lat:.0f}ms     {'✅ 达标' if avg_lat <= 100 else '❌ 未达标'}  (SOP要求 ≤100ms)",
        f"  ≤500ms 达标率:     {under_500:.1f}%",
        f"  vs 基线提升:       +{improvement:.1f}%  (基线={baseline_rate:.1f}% → 算法={accuracy_rate*100:.1f}%)",
        "",
        "【总体请求统计】",
        f"  总请求数:          {total}",
        f"  HTTP成功:          {http_success} ({http_success/total*100:.1f}%)",
        f"  算法识别成功:      {algo_success} ({accuracy_rate*100:.1f}%)",
        f"  超时数:            {timeouts} ({timeouts/total*100:.1f}%)",
        "",
        "【延迟分布】",
        f"  平均: {avg_lat:.0f}ms  P50: {pct(latencies,50):.0f}ms  P90: {pct(latencies,90):.0f}ms  P99: {pct(latencies,99):.0f}ms  最大: {max(latencies) if latencies else 0:.0f}ms",
        f"  ≤100ms 占比: {under_100:.1f}%  |  ≤500ms 占比: {under_500:.1f}%",
        "",
        "【分场景准确率】",
    ]
    for scene, stat in sorted(scene_stats.items()):
        rate = stat["success"] / stat["total"] * 100
        lines.append(f"  {scene:<16} {rate:5.1f}%  ({stat['success']}/{stat['total']})")

    lines += ["", "【失败原因分布】"]
    for reason, count in sorted(fail_reasons.items(), key=lambda x: -x[1]):
        lines.append(f"  {reason:<24} {count}次")

    lines += ["", "【Bad Cases（前10条）】"]
    bad_cases = [r for r in results if not r["is_success"]][:10]
    for i, r in enumerate(bad_cases, 1):
        debug = (r.get("response_json") or {}).get("debug_info", {})
        lines.append(f"  {i:2}. 输入: {r['input'][:35]}")
        lines.append(f"      期望: 起={r['expected_start']} 终={r['expected_end']}")
        lines.append(f"      实际: 起={debug.get('matched_start_place','未识别')} 终={debug.get('matched_end_place','未识别')}")
        lines.append(f"      原因: {r['fail_reason']} | 延迟: {r['latency_ms']:.0f}ms")
        lines.append("")

    lines.append("=" * 62)
    report_text = "\n".join(lines)
    print("\n" + report_text)

    os.makedirs(output_dir, exist_ok=True)
    ts = datetime.now().strftime("%Y%m%d_%H%M%S")

    # 保存 txt
    txt_path = os.path.join(output_dir, f"stress_test_report_{ts}.txt")
    with open(txt_path, "w", encoding="utf-8") as f:
        f.write(report_text)

    # 保存 csv
    csv_path = os.path.join(output_dir, "stress_test_results.csv")
    with open(csv_path, "w", newline="", encoding="utf-8-sig") as f:
        writer = csv.DictWriter(f, fieldnames=[
            "input", "scene", "expected_start", "expected_end",
            "has_nav_intent", "latency_ms", "http_status",
            "is_success", "fail_reason", "match_method_actual"
        ])
        writer.writeheader()
        for r in results:
            writer.writerow({k: r.get(k, "") for k in writer.fieldnames})

    # 保存 json
    json_path = os.path.join(output_dir, f"stress_test_report_{ts}.json")
    summary = {
        "total": total, "algo_success": algo_success,
        "accuracy_rate": round(accuracy_rate, 4),
        "baseline_accuracy_rate": round(baseline_rate/100, 4),
        "improvement_pct": round(improvement, 2),
        "avg_latency_ms": round(avg_lat, 2),
        "p90_latency_ms": round(pct(latencies, 90), 2),
        "sop_accuracy_pass": accuracy_rate >= 0.85,
        "sop_latency_pass": avg_lat <= 100,
        "scene_stats": {k: v for k, v in scene_stats.items()},
        "fail_reasons": dict(fail_reasons),
    }
    with open(json_path, "w", encoding="utf-8") as f:
        json.dump({"summary": summary, "bad_cases": bad_cases[:50]}, f, ensure_ascii=False, indent=2)

    print(f"\n📄 文本报告: {txt_path}")
    print(f"📊 CSV详情: {csv_path}")
    print(f"📦 JSON数据: {json_path}")

    return summary

async def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--url", default="http://127.0.0.1:8000/api/chat")
